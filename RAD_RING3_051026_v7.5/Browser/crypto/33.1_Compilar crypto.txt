# 1. Compilação do código fonte do aplicativo (Com otimização máxima -O3 + SSE3)
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Browser/crypto/crypto.c -o Browser/crypto/crypto.o

# 2. Linkagem estática final utilizando o LD diretamente para controle total (browser.elf)

ld -m elf_x86_64 -T user.ld -static -o crypto.elf \
    system/crt0.o \
    Browser/crypto/crypto.o \
    net_user/socket.o \
