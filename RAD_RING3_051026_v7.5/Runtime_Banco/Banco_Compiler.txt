rm -f Runtime_Banco/core/*.o Runtime_Banco/core_banco/*.o Runtime_Banco/Software/*.o

# ==============================================================================
# LBF OS - SCRIPT DE COMPILAÇÃO DO BANCO DE DADOS NATIVO (RUNTIME_BANCO)
# Estilo tradicional: compila um por um, linka todos no final.
# Gera: Runtime_Banco/lib_banco/banco.o  (+ libbanco.h, que é só header)
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. Compilação dos CORES (as ferramentas), um arquivo por vez
# ------------------------------------------------------------------------------
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/core_banco/bancostore.c  -o Runtime_Banco/core_banco/bancostore.o
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/core_banco/bancoschema.c -o Runtime_Banco/core_banco/bancoschema.o
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/core_banco/bancocreate.c -o Runtime_Banco/core_banco/bancocreate.o
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/core_banco/bancoinsert.c -o Runtime_Banco/core_banco/bancoinsert.o
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/core_banco/bancoselect.c -o Runtime_Banco/core_banco/bancoselect.o
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/core_banco/bancoupdate.c  -o Runtime_Banco/core_banco/bancoupdate.o
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/core_banco/bancodelete.c -o Runtime_Banco/core_banco/bancodelete.o
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/core_banco/bancalter.c -o Runtime_Banco/core_banco/bancalter.o

# ------------------------------------------------------------------------------
# 2. Compilação do DISPATCHER (o divisor de mundos, espelho do neural.c)
# ------------------------------------------------------------------------------
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_Banco/sys_banco/sys_banco.c -o Runtime_Banco/sys_banco/sys_banco.o

# ------------------------------------------------------------------------------
# 3. LINKAGEM PARCIAL (ld -r): dispatcher + todos os cores em UM único objeto.
#    Não usa -T user.ld nem crt0: o resultado é relocável, não executável.
#    É AQUI que nasce o banco.o — o pacote que o software.elf vai linkar.
# ------------------------------------------------------------------------------
ld -r -o Runtime_Banco/lib_banco/banco.o \
    Runtime_Banco/sys_banco/sys_banco.o \
    Runtime_Banco/core_banco/bancostore.o \
    Runtime_Banco/core_banco/bancoschema.o \
    Runtime_Banco/core_banco/bancocreate.o \
    Runtime_Banco/core_banco/bancoinsert.o \
    Runtime_Banco/core_banco/bancoselect.o \
    Runtime_Banco/core_banco/bancoupdate.o \
    Runtime_Banco/core_banco/bancalter.o \
    Runtime_Banco/core_banco/bancodelete.o

# ------------------------------------------------------------------------------
# 4. Conferência rápida (recomendada após cada build)
# ------------------------------------------------------------------------------
nm Runtime_Banco/lib_banco/banco.o | grep " T banco_syscall"   # deve mostrar 1 linha
nm Runtime_Banco/lib_banco/banco.o | grep " U sys_fat"         # dependências externas esperadas
ls -la Runtime_Banco/lib_banco/banco.o
